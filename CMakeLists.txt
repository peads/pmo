cmake_minimum_required(VERSION 3.29)
set(TARGET pmo)
project(${TARGET} CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if (NOT DEFINED GENERATE_TESTS)
    message(AUTHOR_WARNING "This is here to enable testing. The headers should be used directly by your project.")
else ()
    string(TOUPPER "${CMAKE_BUILD_TYPE}" uppercase_CMAKE_BUILD_TYPE)
    if (NOT uppercase_CMAKE_BUILD_TYPE STREQUAL "DEBUG")
        if (CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND MSVC)
            add_compile_options(-Zp8 /clang:-m64 /clang:-O3 /clang:-march=native /WX- /Gy /clang:-flto /clang:-fomit-frame-pointer)
        elseif (MSVC)
            add_compile_options(-Zp8 /O2 /Ox /Ob2 /Ot /Oy /Oi /WX- /GL /Gy /Gm- /MP)
        endif ()
        add_link_options(/OPT:REF /OPT:ICF=3 /LTCG /DEBUG:NONE /EMITPOGOPHASEINFO)
        add_compile_definitions(OPTIMIZATIONS_ON)
    endif()
    add_compile_definitions(WIN32_LEAN_AND_MEAN)
    set(ARE_TESTS_GENERATED ${GENERATE_TESTS}) # cmake throws annoying warning if iT's NoT uSeD
    include(${CMAKE_SOURCE_DIR}/test/tests.cmake)
endif ()
